t.c=o(n)
s.c=o(1)

class Solution {
    public int findMaxConsecutiveOnes(int[] nums) {
         int c1 = 0, c2 = 0;
        for (int x : nums) {
            if (x == 1) {
                c1++;
            } else {
                c2 =Math.max(c1, c2);
                c1 = 0;
            }
        }
        c2 = Math.max(c1,c2);
        return c2;
    }
}
