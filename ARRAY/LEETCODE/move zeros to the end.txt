T.c=o(n),S.c=null
class Solution {
    public void moveZeroes(int[] arr) {
        int n=arr.length;
      int j=0;
      if(n<=1) return;
      for(int i=0;i<n;i++){
        if(arr[i]!=0){
            int temp = arr[i];
            arr[i] = arr[j];
            arr[j] = temp;
            j++;
        }

      }

    }
}
